    64→function pickRandom<T>(arr: T[], n: number): T[] {
    65→  const copy = [...arr];
    66→  const out: T[] = [];
    67→  while (out.length < n && copy.length > 0) {
    68→    out.push(copy.splice(Math.floor(Math.random() * copy.length), 1)[0]);
    69→  }
    70→  return out;
    71→}

    73→/** Приводим распознанную строку к каноническому продукту справочника */
    74→function mapToCanonical(raw: string): string {
    75→  const n = raw.toLowerCase();
    76→  for (const p of FRIDGE_PRODUCTS) {
    77→    if (p.keys.some((k) => n.includes(k))) return p.name;
    78→  }
    79→  return raw;
    80→}

    82→/**
    83→ * Снимок кадра видео в JPEG-датаURL с уменьшением до 1024px.
    84→ * digitalZoom — цифровой зум (CSS fallback): кадр кропается по центру,
    85→ * чтобы снимок совпадал с тем, что видит пользователь.
    86→ */
    87→function frameToDataUrl(video: HTMLVideoElement, digitalZoom: number): string | null {
    88→  if (!video.videoWidth || !video.videoHeight) return null;
    89→  const zoom = Math.max(1, digitalZoom);
    90→  const sw = video.videoWidth / zoom;
    91→  const sh = video.videoHeight / zoom;
    92→  const sx = (video.videoWidth - sw) / 2;
    93→  const sy = (video.videoHeight - sh) / 2;
    94→  const canvas = document.createElement('canvas');
    95→  const scale = Math.min(1, 1024 / Math.max(sw, sh));
    96→  canvas.width = Math.round(sw * scale);
    97→  canvas.height = Math.round(sh * scale);
    98→  const ctx = canvas.getContext('2d');
    99→  if (!ctx) return null;
   100→  ctx.drawImage(video, sx, sy, sw, sh, 0, 0, canvas.width, canvas.height);
   101→  return canvas.toDataURL('image/jpeg', 0.82);
   102→}

   104→/** Файл из галереи → уменьшенный JPEG-датаURL (с учётом EXIF-поворота) */
   105→async function fileToDataUrl(file: File): Promise<string> {
   106→  // createImageBitmap уважает EXIF-поворот — фото с телефона часто повёрнуты,
   107→  // и без этого блюда распознаются «лёжа на боку»
   108→  if (typeof createImageBitmap === 'function') {
   109→    try {
   110→      const bmp = await createImageBitmap(file, { imageOrientation: 'from-image' });
   111→      const canvas = document.createElement('canvas');
   112→      const scale = Math.min(1, 1024 / Math.max(bmp.width, bmp.height));
   113→      canvas.width = Math.round(bmp.width * scale);
   114→      canvas.height = Math.round(bmp.height * scale);
   115→      const ctx = canvas.getContext('2d');
   116→      if (!ctx) throw new Error('canvas');
   117→      ctx.drawImage(bmp, 0, 0, canvas.width, canvas.height);
   118→      bmp.close();
   119→      return canvas.toDataURL('image/jpeg', 0.82);
   120→    } catch {
   121→      /* фолбэк ниже — через Image */
   122→    }
   123→  }
   124→  return new Promise((resolve, reject) => {
   125→    const img = new Image();
   126→    const url = URL.createObjectURL(file);
   127→    img.onload = () => {
   128→      const canvas = document.createElement('canvas');
   129→      const scale = Math.min(1, 1024 / Math.max(img.naturalWidth, img.naturalHeight));
   130→      canvas.width = Math.round(img.naturalWidth * scale);
   131→      canvas.height = Math.round(img.naturalHeight * scale);
   132→      const ctx = canvas.getContext('2d');
   133→      if (!ctx) {
   134→        reject(new Error('canvas'));
   135→        return;
   136→      }
   137→      ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
   138→      URL.revokeObjectURL(url);
   139→      resolve(canvas.toDataURL('image/jpeg', 0.82));
   140→    };
   141→    img.onerror = () => {
   142→      URL.revokeObjectURL(url);
   143→      reject(new Error('image decode'));
   144→    };
   145→    img.src = url;
   146→  });
   147→}

   486→  /** Прямой запрос в Open Food Facts из браузера — фолбэк, когда /api недоступен (статический хостинг) */
   487→  const fetchOffDirect = async (code: string): Promise<BarcodeProduct | null> => {
   488→    const fields =

   490→      'nutriments,ingredients_text,ingredients_text_ru';

   496→    const p = data.product;
   497→    const name: string = (p.product_name_ru || p.product_name || '').trim();
   498→    if (!name) return null;
