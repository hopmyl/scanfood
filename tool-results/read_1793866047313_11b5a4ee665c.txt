    84→function pickRandom<T>(arr: T[], n: number): T[] {
    85→  const copy = [...arr];
    86→  const out: T[] = [];
    87→  while (out.length < n && copy.length > 0) {
    88→    out.push(copy.splice(Math.floor(Math.random() * copy.length), 1)[0]);
    89→  }
    90→  return out;
    91→}

   104→/** Мин зум трека или null, если зум не поддерживается */
   105→function trackZoomMin(track: MediaStreamTrack | null | undefined): number | null {
   106→  if (!track || typeof track.getCapabilities !== 'function') return null;
   107→  try {
   108→    const caps = track.getCapabilities() as unknown as { zoom?: { min?: number; max?: number } };
   109→    if (caps.zoom && typeof caps.zoom.min === 'number') return caps.zoom.min;
   110→  } catch {
   111→    /* некоторые движки кидают на getCapabilities — считаем, что зума нет */
   112→  }
   113→  return null;
   114→}

   116→/** Быстро открыть камеру по id, узнать её минимальный зум и сразу закрыть */
   117→async function probeZoomMin(deviceId: string, timeoutMs = 2500): Promise<number | null> {
   118→  let stream: MediaStream | null = null;
   119→  const stop = () => stream?.getTracks().forEach((t) => t.stop());
   120→  const gump = navigator.mediaDevices
   121→    .getUserMedia({ video: { deviceId: { exact: deviceId } }, audio: false })
   122→    .then((s) => {
   123→      stream = s;
   124→      return s;
   125→    });
   126→  try {
   127→    const s = await Promise.race([
   128→      gump,
   129→      new Promise<never>((_, reject) =>
   130→        setTimeout(() => reject(new Error('probe timeout')), timeoutMs),
   131→      ),
   132→    ]);
   133→    const track = s.getVideoTracks()[0] ?? null;
   134→    // Страховка: вдруг попали на фронталку — её в подбор не берём
   135→    if (track?.getSettings?.().facingMode === 'user') {
   136→      stop();
   137→      return null;
   138→    }
   139→    const min = trackZoomMin(track);
   140→    stop();
   141→    return min;
   142→  } catch {
   143→    // Таймаут: getUserMedia может разрешиться позже — гасим камеру тогда,
   144→    // иначе на медленном телефоне она останется включённой
   145→    void gump.then(stop).catch(stop);
   146→    return null;
   147→  }
   148→}

   150→/**
   151→ * Перебираем «задние» камеры по подписи и ищем самую широкую (min zoom).
   152→ * На десктопе (нет «задних» подписей) и iOS (нет getCapabilities) — no-op.
   153→ */
   154→async function findWidestBackCamera(
   155→  currentDeviceId: string | null,
   156→): Promise<{ deviceId: string; min: number } | null> {
   157→  try {
   158→    const devices = await navigator.mediaDevices.enumerateDevices();
   159→    const isFront = (label: string) => /front|фронт|передн|user|face/i.test(label);
   160→    const backs = devices
   161→      .filter((d) => d.kind === 'videoinput' && d.label && !isFront(d.label))
   162→      .slice(0, 6);
   163→    if (backs.length === 0) return null;
   164→    let best: { deviceId: string; min: number } | null = null;
   165→    for (const d of backs) {
   166→      if (currentDeviceId && d.deviceId === currentDeviceId) continue;
   167→      const min = await probeZoomMin(d.deviceId);
   168→      if (min !== null && min < (best?.min ?? Infinity)) best = { deviceId: d.deviceId, min };
   169→      if (best && best.min <= 0.71) break; // ультраширик найден — хватит
   170→    }
   171→    return best;
   172→  } catch {
   173→    return null;
   174→  }
   175→}

   186→/**
   187→ * Снимок кадра видео в JPEG-датаURL — СТРОГО то, что видит пользователь:
   188→ * сначала кроп под видоискатель 3:4 (object-cover), затем цифровой зум
   189→ * (центральный кроп), затем уменьшение до 1024px. Без этого превью и
   190→ * снимок расходились: пользователь наводил на продукты, а на сервер
   191→ * уходил весь широкий кадр — продукты становились мелкими.
   192→ */
   193→function frameToDataUrl(video: HTMLVideoElement, digitalZoom: number): string | null {
   194→  if (!video.videoWidth || !video.videoHeight) return null;
   195→  const zoom = Math.max(1, digitalZoom);
   196→  // 1) Кроп под видоискатель
   197→  let sw = video.videoWidth;
   198→  let sh = video.videoHeight;
   199→  const streamAR = sw / sh;
   200→  if (streamAR > VIEWFINDER_AR) {
   201→    sw = sh * VIEWFINDER_AR; // альбомный поток — режем бока
   202→  } else if (streamAR < VIEWFINDER_AR) {
   203→    sh = sw / VIEWFINDER_AR; // более вертикальный поток — режем верх/низ
   204→  }
   205→  // 2) Цифровой зум — центральный кроп поверх видоискателя
   206→  sw /= zoom;
   207→  sh /= zoom;
   208→  const sx = (video.videoWidth - sw) / 2;
   209→  const sy = (video.videoHeight - sh) / 2;
   210→  const canvas = document.createElement('canvas');
   211→  const scale = Math.min(1, 1024 / Math.max(sw, sh));
   212→  canvas.width = Math.round(sw * scale);
   213→  canvas.height = Math.round(sh * scale);
   214→  const ctx = canvas.getContext('2d');
   215→  if (!ctx) return null;
   216→  ctx.drawImage(video, sx, sy, sw, sh, 0, 0, canvas.width, canvas.height);
   217→  return canvas.toDataURL('image/jpeg', 0.82);
   218→}

   220→/** Файл из галереи → уменьшенный JPEG-датаURL (с учётом EXIF-поворота) */
   221→async function fileToDataUrl(file: File): Promise<string> {
   222→  // createImageBitmap уважает EXIF-поворот — фото с телефона часто повёрнуты,
   223→  // и без этого блюда распознаются «лёжа на боку»
   224→  if (typeof createImageBitmap === 'function') {
   225→    try {
   226→      const bmp = await createImageBitmap(file, { imageOrientation: 'from-image' });
   227→      const canvas = document.createElement('canvas');
   228→      const scale = Math.min(1, 1024 / Math.max(bmp.width, bmp.height));
   229→      canvas.width = Math.round(bmp.width * scale);
   230→      canvas.height = Math.round(bmp.height * scale);
   231→      const ctx = canvas.getContext('2d');
   232→      if (!ctx) throw new Error('canvas');
   233→      ctx.drawImage(bmp, 0, 0, canvas.width, canvas.height);
   234→      bmp.close();
   235→      return canvas.toDataURL('image/jpeg', 0.82);
   236→    } catch {
   237→      /* фолбэк ниже — через Image */
   238→    }
   239→  }
   240→  return new Promise((resolve, reject) => {
   241→    const drawAndResolve = (
   242→      source: ImageBitmap | HTMLImageElement,
   243→      w: number,
   244→      h: number,
   245→      cleanup: () => void,
   246→    ) => {
   247→      const canvas = document.createElement('canvas');
   248→      const scale = Math.min(1, 1024 / Math.max(w, h));
   249→      canvas.width = Math.round(w * scale);
   250→      canvas.height = Math.round(h * scale);
   251→      const ctx = canvas.getContext('2d');
   252→      if (!ctx) {
   253→        cleanup();
   254→        reject(new Error('canvas'));
   255→        return;
   256→      }
   257→      ctx.drawImage(source, 0, 0, canvas.width, canvas.height);
   258→      cleanup();
   259→      resolve(canvas.toDataURL('image/jpeg', 0.82));
   260→    };
   261→
   262→    // createImageBitmap({ imageOrientation: 'from-image' }) уважает EXIF —
   263→    // фото с телефона не окажется вверх ногами на сервере
   264→    const modern = () =>
   265→      createImageBitmap(file, { imageOrientation: 'from-image' })
   266→        .then((bmp) =>
   267→          drawAndResolve(bmp, bmp.width, bmp.height, () => {
   268→            bmp.close();
   269→          }),
   270→        )
   271→        .catch(() => legacy());
   272→
   273→    const legacy = () => {
   274→      const img = new Image();
   275→      const url = URL.createObjectURL(file);
   276→      img.onload = () =>
   277→        drawAndResolve(img, img.width, img.height, () => URL.revokeObjectURL(url));
   278→      img.onerror = () => {
   279→        URL.revokeObjectURL(url);
   280→        reject(new Error('image decode'));
   281→      };
   282→      img.src = url;
   283→    };
   284→
   285→    if (typeof createImageBitmap === 'function') modern();
   286→    else legacy();
   287→  });
   288→}

   734→    const fields =
   735→      'product_name,product_name_ru,brands,image_front_small_url,image_url,' +
   736→      'nutriments,ingredients_text,ingredients_text_ru';

   758→      brands: (p.brands || '').split(',').map((s: string) => s.trim()).filter(Boolean)[0] ?? '',
