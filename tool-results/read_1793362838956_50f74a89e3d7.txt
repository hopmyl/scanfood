   208→/**
   209→ * Базовые ингредиенты, которые presumed «всегда есть дома»:
   210→ * соль, перец, специи, масло, вода, зелень — не участвуют в подборе рецептов.
   211→ * ВАЖНО: проверка «болгар» идёт первой — болгарский перец это реальный продукт.
   212→ */
   213→export function isBaseIngredient(name: string): boolean {
   214→  const n = name.toLowerCase();
   215→  if (n.includes('болгар')) return false;
   216→  return /(соль|чёрн\S* перец|черн\S* перец|перец, |специ|приправ|масл|вода|мускат|лавров|паприк|зира|орегано|тимьян|розмарин|кунжут|имбирь|васаби|хлопья чили|ванил|укроп|петрушк|кинза|мята|лимонн|порошок|сода|разрыхл|дрожж|уксус|бальзам|кориц|куркум|кориандр|сунели|гвоздик|шафран|прованск|итальянск трав)/.test(
   217→    n,
   218→  );
   219→}
