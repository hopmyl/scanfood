    68→function pickRandom<T>(arr: T[], n: number): T[] {
    69→  const copy = [...arr];
    70→  const out: T[] = [];
    71→  while (out.length < n && copy.length > 0) {
    72→    out.push(copy.splice(Math.floor(Math.random() * copy.length), 1)[0]);
    73→  }
    74→  return out;
    75→}

    88→/** Мин зум трека или null, если зум не поддерживается */
    89→function trackZoomMin(track: MediaStreamTrack | null | undefined): number | null {
    90→  if (!track || typeof track.getCapabilities !== 'function') return null;
    91→  try {
    92→    const caps = track.getCapabilities() as unknown as { zoom?: { min?: number; max?: number } };
    93→    if (caps.zoom && typeof caps.zoom.min === 'number') return caps.zoom.min;
    94→  } catch {
    95→    /* некоторые движки кидают на getCapabilities — считаем, что зума нет */
    96→  }
    97→  return null;
    98→}

   100→/** Быстро открыть камеру по id, узнать её минимальный зум и сразу закрыть */
   101→async function probeZoomMin(deviceId: string, timeoutMs = 2500): Promise<number | null> {
   102→  let stream: MediaStream | null = null;
   103→  const stop = () => stream?.getTracks().forEach((t) => t.stop());
   104→  const gump = navigator.mediaDevices
   105→    .getUserMedia({ video: { deviceId: { exact: deviceId } }, audio: false })
   106→    .then((s) => {
   107→      stream = s;
   108→      return s;
   109→    });
   110→  try {
   111→    const s = await Promise.race([
   112→      gump,
   113→      new Promise<never>((_, reject) =>
   114→        setTimeout(() => reject(new Error('probe timeout')), timeoutMs),
   115→      ),
   116→    ]);
   117→    const track = s.getVideoTracks()[0] ?? null;
   118→    // Страховка: вдруг попали на фронталку — её в подбор не берём
   119→    if (track?.getSettings?.().facingMode === 'user') {
   120→      stop();
   121→      return null;
   122→    }
   123→    const min = trackZoomMin(track);
   124→    stop();
   125→    return min;
   126→  } catch {
   127→    // Таймаут: getUserMedia может разрешиться позже — гасим камеру тогда,
   128→    // иначе на медленном телефоне она останется включённой
   129→    void gump.then(stop).catch(stop);
   130→    return null;
   131→  }
   132→}

   134→/**
   135→ * Перебираем «задние» камеры по подписи и ищем самую широкую (min zoom).
   136→ * На десктопе (нет «задних» подписей) и iOS (нет getCapabilities) — no-op.
   137→ */
   138→async function findWidestBackCamera(
   139→  currentDeviceId: string | null,
   140→): Promise<{ deviceId: string; min: number } | null> {
   141→  try {
   142→    const devices = await navigator.mediaDevices.enumerateDevices();
   143→    const isFront = (label: string) => /front|фронт|передн|user|face/i.test(label);
   144→    const backs = devices
   145→      .filter((d) => d.kind === 'videoinput' && d.label && !isFront(d.label))
   146→      .slice(0, 6);
   147→    if (backs.length === 0) return null;
   148→    let best: { deviceId: string; min: number } | null = null;
   149→    for (const d of backs) {
   150→      if (currentDeviceId && d.deviceId === currentDeviceId) continue;
   151→      const min = await probeZoomMin(d.deviceId);
   152→      if (min !== null && min < (best?.min ?? Infinity)) best = { deviceId: d.deviceId, min };
   153→      if (best && best.min <= 0.71) break; // ультраширик найден — хватит
   154→    }
   155→    return best;
   156→  } catch {
   157→    return null;
   158→  }
   159→}

   170→/**
   171→ * Снимок кадра видео в JPEG-датаURL — СТРОГО то, что видит пользователь:
   172→ * сначала кроп под видоискатель 3:4 (object-cover), затем цифровой зум
   173→ * (центральный кроп), затем уменьшение до 1024px. Без этого превью и
   174→ * снимок расходились: пользователь наводил на продукты, а на сервер
   175→ * уходил весь широкий кадр — продукты становились мелкими.
   176→ */
   177→function frameToDataUrl(video: HTMLVideoElement, digitalZoom: number): string | null {
   178→  if (!video.videoWidth || !video.videoHeight) return null;
   179→  const zoom = Math.max(1, digitalZoom);
   180→  // 1) Кроп под видоискатель
   181→  let sw = video.videoWidth;
   182→  let sh = video.videoHeight;
   183→  const streamAR = sw / sh;
   184→  if (streamAR > VIEWFINDER_AR) {
   185→    sw = sh * VIEWFINDER_AR; // альбомный поток — режем бока
   186→  } else if (streamAR < VIEWFINDER_AR) {
   187→    sh = sw / VIEWFINDER_AR; // более вертикальный поток — режем верх/низ
   188→  }
   189→  // 2) Цифровой зум — центральный кроп поверх видоискателя
   190→  sw /= zoom;
   191→  sh /= zoom;
   192→  const sx = (video.videoWidth - sw) / 2;
   193→  const sy = (video.videoHeight - sh) / 2;
   194→  const canvas = document.createElement('canvas');
   195→  const scale = Math.min(1, 1024 / Math.max(sw, sh));
   196→  canvas.width = Math.round(sw * scale);
   197→  canvas.height = Math.round(sh * scale);
   198→  const ctx = canvas.getContext('2d');
   199→  if (!ctx) return null;
   200→  ctx.drawImage(video, sx, sy, sw, sh, 0, 0, canvas.width, canvas.height);
   201→  return canvas.toDataURL('image/jpeg', 0.82);
   202→}

   204→/** Файл из галереи → уменьшенный JPEG-датаURL (с учётом EXIF-поворота) */
   205→async function fileToDataUrl(file: File): Promise<string> {
   206→  // createImageBitmap уважает EXIF-поворот — фото с телефона часто повёрнуты,
   207→  // и без этого блюда распознаются «лёжа на боку»
   208→  if (typeof createImageBitmap === 'function') {
   209→    try {
   210→      const bmp = await createImageBitmap(file, { imageOrientation: 'from-image' });
   211→      const canvas = document.createElement('canvas');
   212→      const scale = Math.min(1, 1024 / Math.max(bmp.width, bmp.height));
   213→      canvas.width = Math.round(bmp.width * scale);
   214→      canvas.height = Math.round(bmp.height * scale);
   215→      const ctx = canvas.getContext('2d');
   216→      if (!ctx) throw new Error('canvas');
   217→      ctx.drawImage(bmp, 0, 0, canvas.width, canvas.height);
   218→      bmp.close();
   219→      return canvas.toDataURL('image/jpeg', 0.82);
   220→    } catch {
   221→      /* фолбэк ниже — через Image */
   222→    }
   223→  }
   224→  return new Promise((resolve, reject) => {
   225→    const drawAndResolve = (
   226→      source: ImageBitmap | HTMLImageElement,
   227→      w: number,
   228→      h: number,
   229→      cleanup: () => void,
   230→    ) => {
   231→      const canvas = document.createElement('canvas');
   232→      const scale = Math.min(1, 1024 / Math.max(w, h));
   233→      canvas.width = Math.round(w * scale);
   234→      canvas.height = Math.round(h * scale);
   235→      const ctx = canvas.getContext('2d');
   236→      if (!ctx) {
   237→        cleanup();
   238→        reject(new Error('canvas'));
   239→        return;
   240→      }
   241→      ctx.drawImage(source, 0, 0, canvas.width, canvas.height);
   242→      cleanup();
   243→      resolve(canvas.toDataURL('image/jpeg', 0.82));
   244→    };
   245→
   246→    // createImageBitmap({ imageOrientation: 'from-image' }) уважает EXIF —
   247→    // фото с телефона не окажется вверх ногами на сервере
   248→    const modern = () =>
   249→      createImageBitmap(file, { imageOrientation: 'from-image' })
   250→        .then((bmp) =>
   251→          drawAndResolve(bmp, bmp.width, bmp.height, () => {
   252→            bmp.close();
   253→          }),
   254→        )
   255→        .catch(() => legacy());
   256→
   257→    const legacy = () => {
   258→      const img = new Image();
   259→      const url = URL.createObjectURL(file);
   260→      img.onload = () =>
   261→        drawAndResolve(img, img.width, img.height, () => URL.revokeObjectURL(url));
   262→      img.onerror = () => {
   263→        URL.revokeObjectURL(url);
   264→        reject(new Error('image decode'));
   265→      };
   266→      img.src = url;
   267→    };
   268→
   269→    if (typeof createImageBitmap === 'function') modern();
   270→    else legacy();
   271→  });
   272→}
