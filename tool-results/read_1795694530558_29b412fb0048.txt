    20→// Хелперы дат (локальная таймзона, ISO «YYYY-MM-DD»)

    22→
    23→const WD_SHORT = ['Вс', 'Пн', 'Вт', 'Ср', 'Чт', 'Пт', 'Сб'];
    24→
    25→function isoOf(d: Date): string {
    26→  const p = (x: number) => String(x).padStart(2, '0');
    27→  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`;
    28→}
    29→
    30→function parseIso(iso: string): Date {
    31→  const [y, m, d] = iso.split('-').map(Number);
    32→  return new Date(y, m - 1, d);
    33→}
    34→
    35→function addDaysIso(iso: string, n: number): string {
    36→  const d = parseIso(iso);
    37→  d.setDate(d.getDate() + n);
    38→  return isoOf(d);
    39→}
    40→
    41→/** Неделя (Пн-Вс), в которую попадает дата */
    42→function weekOf(iso: string): string[] {
    43→  const d = parseIso(iso);
    44→  const monday = addDaysIso(iso, -((d.getDay() + 6) % 7));
    45→  return Array.from({ length: 7 }, (_, i) => addDaysIso(monday, i));
    46→}
    47→
    48→const wdShort = (iso: string) => WD_SHORT[parseIso(iso).getDay()];
    49→const dayNum = (iso: string) => parseIso(iso).getDate();

    51→/** Слот приёма пищи по текущему часу */
    52→function currentMeal(): MealSlot {
    53→  const h = new Date().getHours();
    54→  if (h < 11) return 'breakfast';
    55→  if (h < 16) return 'lunch';
    56→  if (h < 22) return 'dinner';
    57→  return 'snack';
    58→}
