    50→function pickRandom<T>(arr: T[], n: number): T[] {
    51→  const copy = [...arr];
    52→  const out: T[] = [];
    53→  while (out.length < n && copy.length > 0) {
    54→    out.push(copy.splice(Math.floor(Math.random() * copy.length), 1)[0]);
    55→  }
    56→  return out;
    57→}

    59→/** Приводим распознанную строку к каноническому продукту справочника */
    60→function mapToCanonical(raw: string): string {
    61→  const n = raw.toLowerCase();
    62→  for (const p of FRIDGE_PRODUCTS) {
    63→    if (p.keys.some((k) => n.includes(k))) return p.name;
    64→  }
    65→  return raw;
    66→}

    68→function dedupe(list: string[]): string[] {

    72→/** Снимок кадра видео в JPEG-датаURL с уменьшением до 1024px */
    73→function frameToDataUrl(video: HTMLVideoElement): string | null {
    74→  if (!video.videoWidth || !video.videoHeight) return null;
    75→  const canvas = document.createElement('canvas');
    76→  const scale = Math.min(1, 1024 / Math.max(video.videoWidth, video.videoHeight));
    77→  canvas.width = Math.round(video.videoWidth * scale);
    78→  canvas.height = Math.round(video.videoHeight * scale);
    79→  const ctx = canvas.getContext('2d');
    80→  if (!ctx) return null;
    81→  ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
    82→  return canvas.toDataURL('image/jpeg', 0.82);
    83→}

    85→/** Файл из галереи → уменьшенный JPEG-датаURL */
    86→function fileToDataUrl(file: File): Promise<string> {
    87→  return new Promise((resolve, reject) => {
    88→    const img = new Image();
    89→    const url = URL.createObjectURL(file);
    90→    img.onload = () => {
    91→      const canvas = document.createElement('canvas');
    92→      const scale = Math.min(1, 1024 / Math.max(img.width, img.height));
    93→      canvas.width = Math.round(img.width * scale);
    94→      canvas.height = Math.round(img.height * scale);
    95→      const ctx = canvas.getContext('2d');
    96→      if (!ctx) {
    97→        reject(new Error('canvas'));
    98→        return;
    99→      }
   100→      ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
   101→      URL.revokeObjectURL(url);
   102→      resolve(canvas.toDataURL('image/jpeg', 0.82));
   103→    };
   104→    img.onerror = () => {
   105→      URL.revokeObjectURL(url);
   106→      reject(new Error('image decode'));
   107→    };
   108→    img.src = url;
   109→  });
   110→}
