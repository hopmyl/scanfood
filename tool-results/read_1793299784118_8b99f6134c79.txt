    64→    64→function pickRandom<T>(arr: T[], n: number): T[] {
    65→    65→  const copy = [...arr];
    66→    66→  const out: T[] = [];
    67→    67→  while (out.length < n && copy.length > 0) {
    68→    68→    out.push(copy.splice(Math.floor(Math.random() * copy.length), 1)[0]);
    69→    69→  }
    70→    70→  return out;
    71→    71→}

    82→    82→/**
    83→    83→ * Снимок кадра видео в JPEG-датаURL с уменьшением до 1024px.
    84→    84→ * digitalZoom — цифровой зум (CSS fallback): кадр кропается по центру,
    85→    85→ * чтобы снимок совпадал с тем, что видит пользователь.
    86→    86→ */
    87→    87→function frameToDataUrl(video: HTMLVideoElement, digitalZoom: number): string | null {
    88→    88→  if (!video.videoWidth || !video.videoHeight) return null;
    89→    89→  const zoom = Math.max(1, digitalZoom);
    90→    90→  const sw = video.videoWidth / zoom;
    91→    91→  const sh = video.videoHeight / zoom;
    92→    92→  const sx = (video.videoWidth - sw) / 2;
    93→    93→  const sy = (video.videoHeight - sh) / 2;
    94→    94→  const canvas = document.createElement('canvas');
    95→    95→  const scale = Math.min(1, 1024 / Math.max(sw, sh));
    96→    96→  canvas.width = Math.round(sw * scale);
    97→    97→  canvas.height = Math.round(sh * scale);
    98→    98→  const ctx = canvas.getContext('2d');
    99→    99→  if (!ctx) return null;
   100→   100→  ctx.drawImage(video, sx, sy, sw, sh, 0, 0, canvas.width, canvas.height);
   101→   101→  return canvas.toDataURL('image/jpeg', 0.82);
   102→   102→}

   104→   104→/** Файл из галереи → уменьшенный JPEG-датаURL (с учётом EXIF-поворота) */
   105→   105→async function fileToDataUrl(file: File): Promise<string> {
   106→   106→  // createImageBitmap уважает EXIF-поворот — фото с телефона часто повёрнуты,
   107→   107→  // и без этого блюда распознаются «лёжа на боку»
   108→   108→  if (typeof createImageBitmap === 'function') {
   109→   109→    try {
   110→   110→      const bmp = await createImageBitmap(file, { imageOrientation: 'from-image' });
   111→   111→      const canvas = document.createElement('canvas');
   112→   112→      const scale = Math.min(1, 1024 / Math.max(bmp.width, bmp.height));
   113→   113→      canvas.width = Math.round(bmp.width * scale);
   114→   114→      canvas.height = Math.round(bmp.height * scale);
   115→   115→      const ctx = canvas.getContext('2d');
   116→   116→      if (!ctx) throw new Error('canvas');
   117→   117→      ctx.drawImage(bmp, 0, 0, canvas.width, canvas.height);
   118→   118→      bmp.close();
   119→   119→      return canvas.toDataURL('image/jpeg', 0.82);
   120→   120→    } catch {
   121→   121→      /* фолбэк ниже — через Image */
   122→   122→    }
   123→   123→  }
   124→   124→  return new Promise((resolve, reject) => {
   125→   125→    const img = new Image();
   126→   126→    const url = URL.createObjectURL(file);
   127→   127→    img.onload = () => {
   128→   128→      const canvas = document.createElement('canvas');
   129→   129→      const scale = Math.min(1, 1024 / Math.max(img.naturalWidth, img.naturalHeight));
   130→   130→      canvas.width = Math.round(img.naturalWidth * scale);
   131→   131→      canvas.height = Math.round(img.naturalHeight * scale);
   132→   132→      const ctx = canvas.getContext('2d');
   133→   133→      if (!ctx) {
   134→   134→        reject(new Error('canvas'));
   135→   135→        return;
   136→   136→      }
   137→   137→      ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
   138→   138→      URL.revokeObjectURL(url);
   139→   139→      resolve(canvas.toDataURL('image/jpeg', 0.82));
   140→   140→    };
   141→   141→    img.onerror = () => {
   142→   142→      URL.revokeObjectURL(url);
   143→   143→      reject(new Error('image decode'));
   144→   144→    };
   145→   145→    img.src = url;
   146→   146→  });
   147→   147→}
