    52→function pickRandom<T>(arr: T[], n: number): T[] {
    53→  const copy = [...arr];
    54→  const out: T[] = [];
    55→  while (out.length < n && copy.length > 0) {
    56→    out.push(copy.splice(Math.floor(Math.random() * copy.length), 1)[0]);
    57→  }
    58→  return out;
    59→}

    61→/** Приводим распознанную строку к каноническому продукту справочника */
    62→function mapToCanonical(raw: string): string {
    63→  const n = raw.toLowerCase();
    64→  for (const p of FRIDGE_PRODUCTS) {
    65→    if (p.keys.some((k) => n.includes(k))) return p.name;
    66→  }
    67→  return raw;
    68→}

    70→function dedupe(list: string[]): string[] {

    74→/** Снимок кадра видео в JPEG-датаURL с уменьшением до 1024px */
    75→function frameToDataUrl(video: HTMLVideoElement): string | null {
    76→  if (!video.videoWidth || !video.videoHeight) return null;
    77→  const canvas = document.createElement('canvas');
    78→  const scale = Math.min(1, 1024 / Math.max(video.videoWidth, video.videoHeight));
    79→  canvas.width = Math.round(video.videoWidth * scale);
    80→  canvas.height = Math.round(video.videoHeight * scale);
    81→  const ctx = canvas.getContext('2d');
    82→  if (!ctx) return null;
    83→  ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
    84→  return canvas.toDataURL('image/jpeg', 0.82);
    85→}

    87→/** Файл из галереи → уменьшенный JPEG-датаURL */
    88→function fileToDataUrl(file: File): Promise<string> {
    89→  return new Promise((resolve, reject) => {
    90→    const img = new Image();
    91→    const url = URL.createObjectURL(file);
    92→    img.onload = () => {
    93→      const canvas = document.createElement('canvas');
    94→      const scale = Math.min(1, 1024 / Math.max(img.width, img.height));
    95→      canvas.width = Math.round(img.width * scale);
    96→      canvas.height = Math.round(img.height * scale);
    97→      const ctx = canvas.getContext('2d');
    98→      if (!ctx) {
    99→        reject(new Error('canvas'));
   100→        return;
   101→      }
   102→      ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
   103→      URL.revokeObjectURL(url);
   104→      resolve(canvas.toDataURL('image/jpeg', 0.82));
   105→    };
   106→    img.onerror = () => {
   107→      URL.revokeObjectURL(url);
   108→      reject(new Error('image decode'));
   109→    };
   110→    img.src = url;
   111→  });
   112→}
