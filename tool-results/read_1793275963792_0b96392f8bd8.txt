    56→function pickRandom<T>(arr: T[], n: number): T[] {
    57→  const copy = [...arr];
    58→  const out: T[] = [];
    59→  while (out.length < n && copy.length > 0) {
    60→    out.push(copy.splice(Math.floor(Math.random() * copy.length), 1)[0]);
    61→  }
    62→  return out;
    63→}

    65→/** Приводим распознанную строку к каноническому продукту справочника */
    66→function mapToCanonical(raw: string): string {
    67→  const n = raw.toLowerCase();
    68→  for (const p of FRIDGE_PRODUCTS) {
    69→    if (p.keys.some((k) => n.includes(k))) return p.name;
    70→  }
    71→  return raw;
    72→}

    78→/**
    79→ * Снимок кадра видео в JPEG-датаURL с уменьшением до 1024px.
    80→ * digitalZoom — цифровой зум (CSS fallback): кадр кропается по центру,
    81→ * чтобы снимок совпадал с тем, что видит пользователь.
    82→ */
    83→function frameToDataUrl(video: HTMLVideoElement, digitalZoom: number): string | null {
    84→  if (!video.videoWidth || !video.videoHeight) return null;
    85→  const zoom = Math.max(1, digitalZoom);
    86→  const sw = video.videoWidth / zoom;
    87→  const sh = video.videoHeight / zoom;
    88→  const sx = (video.videoWidth - sw) / 2;
    89→  const sy = (video.videoHeight - sh) / 2;
    90→  const canvas = document.createElement('canvas');
    91→  const scale = Math.min(1, 1024 / Math.max(sw, sh));
    92→  canvas.width = Math.round(sw * scale);
    93→  canvas.height = Math.round(sh * scale);
    94→  const ctx = canvas.getContext('2d');
    95→  if (!ctx) return null;
    96→  ctx.drawImage(video, sx, sy, sw, sh, 0, 0, canvas.width, canvas.height);
    97→  return canvas.toDataURL('image/jpeg', 0.82);
    98→}

   100→/** Файл из галереи → уменьшенный JPEG-датаURL */
   101→function fileToDataUrl(file: File): Promise<string> {
   102→  return new Promise((resolve, reject) => {
   103→    const img = new Image();
   104→    const url = URL.createObjectURL(file);
   105→    img.onload = () => {
   106→      const canvas = document.createElement('canvas');
   107→      const scale = Math.min(1, 1024 / Math.max(img.width, img.height));
   108→      canvas.width = Math.round(img.width * scale);
   109→      canvas.height = Math.round(img.height * scale);
   110→      const ctx = canvas.getContext('2d');
   111→      if (!ctx) {
   112→        reject(new Error('canvas'));
   113→        return;
   114→      }
   115→      ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
   116→      URL.revokeObjectURL(url);
   117→      resolve(canvas.toDataURL('image/jpeg', 0.82));
   118→    };
   119→    img.onerror = () => {
   120→      URL.revokeObjectURL(url);
   121→      reject(new Error('image decode'));
   122→    };
   123→    img.src = url;
   124→  });
   125→}
