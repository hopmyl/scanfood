    70→function pickRandom<T>(arr: T[], n: number): T[] {
    71→  const copy = [...arr];
    72→  const out: T[] = [];
    73→  while (out.length < n && copy.length > 0) {
    74→    out.push(copy.splice(Math.floor(Math.random() * copy.length), 1)[0]);
    75→  }
    76→  return out;
    77→}

    90→/** Мин зум трека или null, если зум не поддерживается */
    91→function trackZoomMin(track: MediaStreamTrack | null | undefined): number | null {
    92→  if (!track || typeof track.getCapabilities !== 'function') return null;
    93→  try {
    94→    const caps = track.getCapabilities() as unknown as { zoom?: { min?: number; max?: number } };
    95→    if (caps.zoom && typeof caps.zoom.min === 'number') return caps.zoom.min;
    96→  } catch {
    97→    /* некоторые движки кидают на getCapabilities — считаем, что зума нет */
    98→  }
    99→  return null;
   100→}

   102→/** Быстро открыть камеру по id, узнать её минимальный зум и сразу закрыть */
   103→async function probeZoomMin(deviceId: string, timeoutMs = 2500): Promise<number | null> {
   104→  let stream: MediaStream | null = null;
   105→  const stop = () => stream?.getTracks().forEach((t) => t.stop());
   106→  const gump = navigator.mediaDevices
   107→    .getUserMedia({ video: { deviceId: { exact: deviceId } }, audio: false })
   108→    .then((s) => {
   109→      stream = s;
   110→      return s;
   111→    });
   112→  try {
   113→    const s = await Promise.race([
   114→      gump,
   115→      new Promise<never>((_, reject) =>
   116→        setTimeout(() => reject(new Error('probe timeout')), timeoutMs),
   117→      ),
   118→    ]);
   119→    const track = s.getVideoTracks()[0] ?? null;
   120→    // Страховка: вдруг попали на фронталку — её в подбор не берём
   121→    if (track?.getSettings?.().facingMode === 'user') {
   122→      stop();
   123→      return null;
   124→    }
   125→    const min = trackZoomMin(track);
   126→    stop();
   127→    return min;
   128→  } catch {
   129→    // Таймаут: getUserMedia может разрешиться позже — гасим камеру тогда,
   130→    // иначе на медленном телефоне она останется включённой
   131→    void gump.then(stop).catch(stop);
   132→    return null;
   133→  }
   134→}

   136→/**
   137→ * Перебираем «задние» камеры по подписи и ищем самую широкую (min zoom).
   138→ * На десктопе (нет «задних» подписей) и iOS (нет getCapabilities) — no-op.
   139→ */
   140→async function findWidestBackCamera(
   141→  currentDeviceId: string | null,
   142→): Promise<{ deviceId: string; min: number } | null> {
   143→  try {
   144→    const devices = await navigator.mediaDevices.enumerateDevices();
   145→    const isFront = (label: string) => /front|фронт|передн|user|face/i.test(label);
   146→    const backs = devices
   147→      .filter((d) => d.kind === 'videoinput' && d.label && !isFront(d.label))
   148→      .slice(0, 6);
   149→    if (backs.length === 0) return null;
   150→    let best: { deviceId: string; min: number } | null = null;
   151→    for (const d of backs) {
   152→      if (currentDeviceId && d.deviceId === currentDeviceId) continue;
   153→      const min = await probeZoomMin(d.deviceId);
   154→      if (min !== null && min < (best?.min ?? Infinity)) best = { deviceId: d.deviceId, min };
   155→      if (best && best.min <= 0.71) break; // ультраширик найден — хватит
   156→    }
   157→    return best;
   158→  } catch {
   159→    return null;
   160→  }
   161→}

   172→/**
   173→ * Снимок кадра видео в JPEG-датаURL — СТРОГО то, что видит пользователь:
   174→ * сначала кроп под видоискатель 3:4 (object-cover), затем цифровой зум
   175→ * (центральный кроп), затем уменьшение до 1024px. Без этого превью и
   176→ * снимок расходились: пользователь наводил на продукты, а на сервер
   177→ * уходил весь широкий кадр — продукты становились мелкими.
   178→ */
   179→function frameToDataUrl(video: HTMLVideoElement, digitalZoom: number): string | null {
   180→  if (!video.videoWidth || !video.videoHeight) return null;
   181→  const zoom = Math.max(1, digitalZoom);
   182→  // 1) Кроп под видоискатель
   183→  let sw = video.videoWidth;
   184→  let sh = video.videoHeight;
   185→  const streamAR = sw / sh;
   186→  if (streamAR > VIEWFINDER_AR) {
   187→    sw = sh * VIEWFINDER_AR; // альбомный поток — режем бока
   188→  } else if (streamAR < VIEWFINDER_AR) {
   189→    sh = sw / VIEWFINDER_AR; // более вертикальный поток — режем верх/низ
   190→  }
   191→  // 2) Цифровой зум — центральный кроп поверх видоискателя
   192→  sw /= zoom;
   193→  sh /= zoom;
   194→  const sx = (video.videoWidth - sw) / 2;
   195→  const sy = (video.videoHeight - sh) / 2;
   196→  const canvas = document.createElement('canvas');
   197→  const scale = Math.min(1, 1024 / Math.max(sw, sh));
   198→  canvas.width = Math.round(sw * scale);
   199→  canvas.height = Math.round(sh * scale);
   200→  const ctx = canvas.getContext('2d');
   201→  if (!ctx) return null;
   202→  ctx.drawImage(video, sx, sy, sw, sh, 0, 0, canvas.width, canvas.height);
   203→  return canvas.toDataURL('image/jpeg', 0.82);
   204→}

   206→/** Файл из галереи → уменьшенный JPEG-датаURL (с учётом EXIF-поворота) */
   207→async function fileToDataUrl(file: File): Promise<string> {
   208→  // createImageBitmap уважает EXIF-поворот — фото с телефона часто повёрнуты,
   209→  // и без этого блюда распознаются «лёжа на боку»
   210→  if (typeof createImageBitmap === 'function') {
   211→    try {
   212→      const bmp = await createImageBitmap(file, { imageOrientation: 'from-image' });
   213→      const canvas = document.createElement('canvas');
   214→      const scale = Math.min(1, 1024 / Math.max(bmp.width, bmp.height));
   215→      canvas.width = Math.round(bmp.width * scale);
   216→      canvas.height = Math.round(bmp.height * scale);
   217→      const ctx = canvas.getContext('2d');
   218→      if (!ctx) throw new Error('canvas');
   219→      ctx.drawImage(bmp, 0, 0, canvas.width, canvas.height);
   220→      bmp.close();
   221→      return canvas.toDataURL('image/jpeg', 0.82);
   222→    } catch {
   223→      /* фолбэк ниже — через Image */
   224→    }
   225→  }
   226→  return new Promise((resolve, reject) => {
   227→    const drawAndResolve = (
   228→      source: ImageBitmap | HTMLImageElement,
   229→      w: number,
   230→      h: number,
   231→      cleanup: () => void,
   232→    ) => {
   233→      const canvas = document.createElement('canvas');
   234→      const scale = Math.min(1, 1024 / Math.max(w, h));
   235→      canvas.width = Math.round(w * scale);
   236→      canvas.height = Math.round(h * scale);
   237→      const ctx = canvas.getContext('2d');
   238→      if (!ctx) {
   239→        cleanup();
   240→        reject(new Error('canvas'));
   241→        return;
   242→      }
   243→      ctx.drawImage(source, 0, 0, canvas.width, canvas.height);
   244→      cleanup();
   245→      resolve(canvas.toDataURL('image/jpeg', 0.82));
   246→    };
   247→
   248→    // createImageBitmap({ imageOrientation: 'from-image' }) уважает EXIF —
   249→    // фото с телефона не окажется вверх ногами на сервере
   250→    const modern = () =>
   251→      createImageBitmap(file, { imageOrientation: 'from-image' })
   252→        .then((bmp) =>
   253→          drawAndResolve(bmp, bmp.width, bmp.height, () => {
   254→            bmp.close();
   255→          }),
   256→        )
   257→        .catch(() => legacy());
   258→
   259→    const legacy = () => {
   260→      const img = new Image();
   261→      const url = URL.createObjectURL(file);
   262→      img.onload = () =>
   263→        drawAndResolve(img, img.width, img.height, () => URL.revokeObjectURL(url));
   264→      img.onerror = () => {
   265→        URL.revokeObjectURL(url);
   266→        reject(new Error('image decode'));
   267→      };
   268→      img.src = url;
   269→    };
   270→
   271→    if (typeof createImageBitmap === 'function') modern();
   272→    else legacy();
   273→  });
   274→}

   744→      brands: (p.brands || '').split(',').map((s: string) => s.trim()).filter(Boolean)[0] ?? '',
