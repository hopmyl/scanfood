    23→    23→const WD_SHORT = ['Вс', 'Пн', 'Вт', 'Ср', 'Чт', 'Пт', 'Сб'];
    24→    24→
    25→    25→function isoOf(d: Date): string {
    26→    26→  const p = (x: number) => String(x).padStart(2, '0');
    27→    27→  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`;
    28→    28→}
    29→    29→
    30→    30→function parseIso(iso: string): Date {
    31→    31→  const [y, m, d] = iso.split('-').map(Number);
    32→    32→  return new Date(y, m - 1, d);
    33→    33→}
    34→    34→
    35→    35→function addDaysIso(iso: string, n: number): string {
    36→    36→  const d = parseIso(iso);
    37→    37→  d.setDate(d.getDate() + n);
    38→    38→  return isoOf(d);
    39→    39→}
    40→    40→
    41→    41→/** Неделя (Пн-Вс), в которую попадает дата */
    42→    42→function weekOf(iso: string): string[] {
    43→    43→  const d = parseIso(iso);
    44→    44→  const monday = addDaysIso(iso, -((d.getDay() + 6) % 7));
    45→    45→  return Array.from({ length: 7 }, (_, i) => addDaysIso(monday, i));
    46→    46→}
