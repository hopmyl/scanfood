   194→/**
   195→ * Базовые ингредиенты, которые presumed «всегда есть дома»:
   196→ * соль, перец, специи, масло, вода, зелень — не участвуют в подборе рецептов.
   197→ * ВАЖНО: проверка «болгар» идёт первой — болгарский перец это реальный продукт.
   198→ */
   199→export function isBaseIngredient(name: string): boolean {
   200→  const n = name.toLowerCase();
   201→  if (n.includes('болгар')) return false;
   202→  return /(соль|чёрн\S* перец|черн\S* перец|перец, |специ|приправ|масл|вода|мускат|лавров|паприк|зира|орегано|тимьян|розмарин|кунжут|имбирь|васаби|хлопья чили|ванил|укроп|петрушк|кинза|мята|лимонн|порошок|сода|разрыхл|дрожж|уксус|бальзам|кориц|куркум|кориандр|сунели|гвоздик|шафран|прованск|итальянск трав)/.test(
   203→    n,
   204→  );
   205→}
